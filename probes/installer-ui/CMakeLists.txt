cmake_minimum_required(VERSION 3.20)
project(swcompat_installer_ui_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Compile genuine Win32 window census on Windows only.")
endif()
add_executable(solidworks_installer_ui_probe ../solidworks_installer_ui_probe.cpp)
target_compile_features(solidworks_installer_ui_probe PRIVATE cxx_std_17)
target_link_libraries(solidworks_installer_ui_probe PRIVATE user32 kernel32 gdi32)
if(MSVC)
  target_compile_options(solidworks_installer_ui_probe PRIVATE /W4 /EHsc /utf-8)
endif()
